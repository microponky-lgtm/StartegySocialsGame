function playTone(freq, dur, type = 'sine', gain = 0.16, delay = 0, slideTo = null) {
  if (!soundOn) return;
  try {
    audioCtx = audioCtx || new (window.AudioContext || window.webkitAudioContext)();
    if (audioCtx.state === 'suspended') audioCtx.resume();
    const t = audioCtx.currentTime + delay;
    const osc = audioCtx.createOscillator();
    const g = audioCtx.createGain();
    osc.type = type;
    osc.frequency.setValueAtTime(freq, t);
    if (slideTo) osc.frequency.exponentialRampToValueAtTime(slideTo, t + dur);
    g.gain.setValueAtTime(0.0001, t);
    g.gain.exponentialRampToValueAtTime(gain, t + 0.015);
    g.gain.exponentialRampToValueAtTime(0.0001, t + dur);
    osc.connect(g);
    g.connect(audioCtx.destination);
    osc.start(t);
    osc.stop(t + dur + 0.03);
  } catch (e) {}
}

function copyText(text, label = 'Copied!') {
  if (navigator.clipboard && navigator.clipboard.writeText) {
    navigator.clipboard.writeText(text).then(() => toast(label)).catch(() => fallbackCopy(text, label));
  } else {
    fallbackCopy(text, label);
  }
}
function fallbackCopy(text, label) {
  const ta = document.createElement('textarea');
  ta.value = text;
  ta.style.position = 'fixed';
  ta.style.opacity = '0';
  document.body.appendChild(ta);
  ta.select();
  try { document.execCommand('copy'); toast(label); } catch (e) { toast('Copy failed — select manually'); }
  ta.remove();
}

function compressImage(file) {
  return new Promise((resolve, reject) => {
    const img = new Image();
    const url = URL.createObjectURL(file);
    img.onload = () => {
      const scale = Math.min(1, 1080 / Math.max(img.width, img.height));
      const c = document.createElement('canvas');
      c.width = Math.round(img.width * scale);
      c.height = Math.round(img.height * scale);
      c.getContext('2d').drawImage(img, 0, 0, c.width, c.height);
      URL.revokeObjectURL(url);
      resolve(c.toDataURL('image/jpeg', 0.72));
    };
    img.onerror = err => { URL.revokeObjectURL(url); reject(err); };
    img.src = url;
  });
}

function importQuizPack(file) {
  if (!file) return;
  const r = new FileReader();
  r.onload = () => {
    try {
      const parsed = JSON.parse(r.result);
      if (!Array.isArray(parsed.questions)) throw Error('Invalid pack');
      setupState = { ...setupState, ...parsed, roomCode: makeCode() };
      renderHostSetup();
      toast(`Loaded ${setupState.questions.length} questions!`);
    } catch (e) {
      toast('Invalid quiz pack JSON file.');
    }
  };
  r.readAsText(file);
}

  };

    roomId = code;

/* ============================================================================
   FIREBASE REALTIME LISTENERS
   ============================================================================ */
function detachListeners() {
  listeners.forEach(([ref, fn]) => ref.off('value', fn));
  listeners = [];
}

function listenFirebaseRoom() {
  detachListeners();
  renderedLobbyPlayers.clear();
  lbRowNodes = {};
  lastScreenKey = '';
  const base = db.ref('rooms/' + roomId);
  ['meta', 'state', 'questions', 'players', 'buzz'].forEach(key => {
    const r = base.child(key);
    const fn = snap => {
      roomData[key] = snap.val() || (key === 'questions' ? [] : {});
      if (key === 'meta') applyTheme(roomData.meta.theme);
      handleReactiveAudio();
      renderRoom();
    };
    r.on('value', fn);
    listeners.push([r, fn]);
  });
}

function handleReactiveAudio() {
  const s = roomData.state || {};
  const active = getActiveBuzzer();
  const activeId = active ? active[0] : '';
  const outCount = getBuzzQueue().filter(([, b]) => b.out).length;
  const pCount = Object.keys(roomData.players || {}).length;

  if (s.phase === 'lobby' && pCount > seenState.playerCount && seenState.playerCount > 0) {
    playSound('join');
  }
  if (s.phase === 'question' && (seenState.phase !== 'question' || seenState.questionIndex !== s.questionIndex)) {
    playSound('start');
  }
  if (s.phase === 'question' && activeId && activeId !== seenState.activeBuzzId) {
    playSound('buzz');
    if (role === 'host') toast(`${active[1].name} buzzed in!`);
  }
  if (s.phase === 'question' && outCount > seenState.outCount) {
    playSound('wrong');
  }
  if (s.phase === 'reveal' && seenState.phase !== 'reveal') {
    if (s.wasCorrect) { playSound('correct'); launchConfetti(1); }
    else playSound('reveal');
  }
  if (s.phase === 'leaderboard' && seenState.phase !== 'leaderboard') {
    playSound('leaderboard');
  }
  if (s.phase === 'end' && seenState.phase !== 'end') {
    playSound('finale');
    launchConfetti(1.8);
  }

  seenState = {
    phase: s.phase || '',
    activeBuzzId: activeId,
    outCount,
    playerCount: pCount,
    questionIndex: s.questionIndex ?? -1
  };
}

async function playerBuzz() {
  const s = roomData.state || {};
  if (s.phase !== 'question') return;
  if ((roomData.buzz || {})[playerId]) return;
  const me = (roomData.players || {})[playerId];
  if (!me) return;
  try { navigator.vibrate?.(45); } catch (e) {}

  if (IS_LIVE_FIREBASE) {
    await db.ref(`rooms/${roomId}/buzz/${playerId}`).set({
      name: me.name,
      avatar: me.avatar || '⚡',
      t: firebase.database.ServerValue.TIMESTAMP,
      out: false
    });
  } else {
    roomData.buzz[playerId] = { name: me.name, avatar: me.avatar || '⚡', t: Date.now(), out: false };
    handleReactiveAudio();
    renderRoom();
  }
}

let judgeLock = false;
async function judgeActive(isCorrect) {
  if (judgeLock) return;
  const active = getActiveBuzzer();
  if (!active) return;
  judgeLock = true;
  const [pid, b] = active;
  const s = roomData.state;
  const q = (roomData.questions || [])[s.questionIndex] || {};
  const pts = Number(roomData.meta.points) || 10;
  const pen = Number(roomData.meta.wrongPenalty) || 0;

  try {
    if (isCorrect) {
      if (IS_LIVE_FIREBASE) {
        await db.ref(`rooms/${roomId}/players/${pid}/score`).transaction(cur => (cur || 0) + pts);
        await db.ref(`rooms/${roomId}/players/${pid}/wins`).transaction(cur => (cur || 0) + 1);
        await db.ref(`rooms/${roomId}/state`).update({
          phase: 'reveal',
          answer: q.answer || '',
          awardedTo: b.name,
          awardedAvatar: b.avatar || '🏆',
          wasCorrect: true,
          lastWrong: ''
        });
      } else {
        const p = roomData.players[pid];
        if (p) { p.score = (p.score || 0) + pts; p.wins = (p.wins || 0) + 1; }
        Object.assign(roomData.state, {
          phase: 'reveal', answer: q.answer || '', awardedTo: b.name, awardedAvatar: b.avatar || '🏆', wasCorrect: true, lastWrong: ''
        });
        handleReactiveAudio();
        renderRoom();
      }
    } else {
      if (IS_LIVE_FIREBASE) {
        if (pen > 0) {
          await db.ref(`rooms/${roomId}/players/${pid}/score`).transaction(cur => Math.max(0, (cur || 0) - pen));
        }
        await db.ref(`rooms/${roomId}/buzz/${pid}/out`).set(true);
        await db.ref(`rooms/${roomId}/state/lastWrong`).set(b.name);
      } else {
        if (pen > 0 && roomData.players[pid]) {
          roomData.players[pid].score = Math.max(0, (roomData.players[pid].score || 0) - pen);
        }
        roomData.buzz[pid].out = true;
        roomData.state.lastWrong = b.name;
        handleReactiveAudio();
        renderRoom();
      }
    }
  } finally {
    judgeLock = false;
  }
}

async function nextQuestion() {
  const s = roomData.state;
  const list = getSortedPlayers();
  list.forEach(([id], idx) => { prevRankMap[id] = idx; });

  if (s.questionIndex + 1 >= s.total) {
    await updateState({ phase: 'end' });
    return;
  }
  if (IS_LIVE_FIREBASE) {
    await db.ref(`rooms/${roomId}`).update({
      buzz: null,
      'state/phase': 'question',
      'state/questionIndex': s.questionIndex + 1,
      'state/answer': '',
      'state/awardedTo': '',
      'state/awardedAvatar': '',
      'state/wasCorrect': false,
      'state/lastWrong': ''
    });
  } else {
    roomData.buzz = {};
    Object.assign(roomData.state, {
      phase: 'question', questionIndex: s.questionIndex + 1, answer: '', awardedTo: '', awardedAvatar: '', wasCorrect: false, lastWrong: ''
    });
    handleReactiveAudio();
    renderRoom();
  }
}

async function continueFromLeaderboard() {
  const s = roomData.state;
  if (s.returnPhase === 'reveal') {
    await nextQuestion();
  } else {
    await updateState({ phase: s.returnPhase || 'question' });
  }
}

  renderRoom();
}

function demoSimBuzz() {
  if (roomData.state.phase !== 'question') return;
  const candidates = Object.entries(roomData.players || {}).filter(([id]) => !(roomData.buzz || {})[id]);
  if (!candidates.length) return toast('Everyone has already buzzed!');
  const [id, p] = candidates[Math.floor(Math.random() * candidates.length)];
  roomData.buzz[id] = { name: p.name, avatar: p.avatar || '⚡', t: Date.now(), out: false };
  handleReactiveAudio();
  renderRoom();
}

/* ============================================================================
   RENDER DISPATCHER (Zero-Flicker Lobby & Smooth Animated Leaderboard)
   ============================================================================ */
function renderRoom() {
  const s = roomData.state || {};
  applyTheme(roomData.meta?.theme || 'indigo');
  if (s.phase === 'lobby') return renderLobby();
  if (s.phase === 'question') return renderQuestion();
  if (s.phase === 'reveal') return renderReveal();
  if (s.phase === 'leaderboard') return renderLeaderboard();
  if (s.phase === 'end') return renderEnd();
}

function renderQuestion() {
  lastScreenKey = 'question';
  const s = roomData.state || {};
  const q = (roomData.questions || [])[s.questionIndex] || {};
  const isHost = role === 'host';
  const queue = getBuzzQueue();
  const active = getActiveBuzzer();
  const pct = s.total ? Math.round(((s.questionIndex + 1) / s.total) * 100) : 0;

  if (!isHost) {
    // PHONE PLAYER BUZZER VIEW
    const me = (roomData.players || {})[playerId] || { name: 'Player', avatar: '⚡', score: 0 };
    const myBuzz = (roomData.buzz || {})[playerId];
    const activeQueue = queue.filter(([, b]) => !b.out);
    const myPos = activeQueue.findIndex(([id]) => id === playerId) + 1;

    let btnClass = 'big-buzzer';
    let btnLabel = 'BUZZ!';
    let subText = 'Tap as fast as you can when you know the answer!';
    let disabled = false;

    if (myBuzz) {
      disabled = true;
      if (myBuzz.out) {
        btnClass += ' wrong-out';
        btnLabel = 'OUT ❌';
        subText = 'Wrong answer — wait for the next round!';
      } else if (myPos === 1) {
        btnClass += ' first-up';
        btnLabel = 'YOU\'RE #1!';
        subText = 'Say your answer out loud now!';
      } else {
        btnClass += ' queued';
        btnLabel = `QUEUE #${myPos}`;
        subText = 'You are locked in! If players ahead miss, you are up next.';
      }
    }

    appEl.innerHTML = `
      <div class="shell">
        ${headerHTML()}
        <div class="phone-stage">
          <div class="player-mini-bar">
            <div style="display:flex;align-items:center;gap:10px">
              <span style="font-size:22px">${esc(me.avatar || '⚡')}</span>
              <b>${esc(me.name)}</b>
            </div>
            <div style="font-family:'Space Grotesk';font-weight:700;color:var(--gold)">${me.score || 0} pts</div>
          </div>

          <div class="eyebrow">ROUND ${s.questionIndex + 1} OF ${s.total}</div>
          ${q.prompt ? `<div style="font-weight:700;font-size:18px;margin:6px 0 12px">${esc(q.prompt)}</div>` : ''}

            <button class="${btnClass}" onclick="playerBuzz()" ${disabled ? 'disabled' : ''}>${btnLabel}</button>
            <span class="buzz-ring"></span>
          </div>

          <p class="muted" style="font-weight:600">${subText}</p>
          ${active && active[0] !== playerId ? `<div class="queue-pill" style="display:inline-block;margin-top:14px">🔔 Answering now: <b>${esc(active[1].name)}</b></div>` : ''}
        </div>
      </div>`;
    return;
  }

  // HOST / PROJECTOR STAGE VIEW
  const waitingQueue = queue.filter(([id, b]) => !b.out && (!active || id !== active[0]));
  const outPlayers = queue.filter(([, b]) => b.out);

  appEl.innerHTML = `
    <div class="shell">
      ${headerHTML()}
      <div class="q-progress-wrap">
        <span class="eyebrow">ROUND ${s.questionIndex + 1} / ${s.total}</span>
        <div class="q-progress"><i style="width:${pct}%"></i></div>
        <span class="tiny">${Object.keys(roomData.players || {}).length} players</span>
      </div>

      <div class="stage-card">
        ${q.prompt ? `<div class="q-prompt-text">${esc(q.prompt)}</div>` : ''}
        ${q.image ? `<div class="q-image-frame"><img class="q-image" src="${q.image}" alt="Question ${s.questionIndex + 1}"></div>` : ''}

        ${active ? `
          <div class="buzz-spotlight">
            <div class="buzz-spotlight-avatar">${esc(active[1].avatar || '🔔')}</div>
            <div style="text-align:left">
              <div class="eyebrow" style="color:#fde68a">🔔 BUZZED IN FIRST!</div>
              <div class="buzz-spotlight-name">${esc(active[1].name)}</div>
            </div>
          </div>` : `
          <div style="margin-top:20px">
            <div style="font-family:'Space Grotesk';font-weight:700;font-size:clamp(22px,3vw,34px)">Waiting for a buzzer…</div>
            <div class="tiny" style="margin-top:4px">Everyone can press BUZZ on their phone — fastest finger gets the mic!</div>
          </div>`}

          <div class="buzz-queue-strip">
            ${waitingQueue.map(([, b], idx) => `<span class="queue-pill">Next #${idx + 2}: ${esc(b.avatar || '')} ${esc(b.name)}</span>`).join('')}

          </div>` : ''}

        <div class="host-dock">
          ${active ? `

        </div>
      </div>
    </div>`;
}

function renderLeaderboard() {
  const list = getSortedPlayers();
  const s = roomData.state || {};
  const m = roomData.meta || {};
  const isHost = role === 'host';
  const isLast = s.questionIndex + 1 >= s.total;
  const key = 'leaderboard_' + role;
  const ROW_H = 78;

  if (lastScreenKey !== key) {
    lastScreenKey = key;
    lbRowNodes = {};
    appEl.innerHTML = `
      <div class="shell">
        ${headerHTML()}
        <div class="lb-wrap">
          <div class="eyebrow">LIVE STANDINGS</div>
          <h1 class="h1-hero" style="font-size:clamp(34px,5.5vw,58px);margin:8px 0 4px">Leaderboard<span class="grad-text">.</span></h1>
          <p class="muted">${m.rewardEmoji ? `${m.rewardEmoji} = ${m.rewardName || 'Rewards'} won · ` : ''}Watch the ranks climb live!</p>
          <div id="lbTrack" class="lb-track"></div>
          <div style="display:flex;gap:12px;justify-content:center;flex-wrap:wrap;margin-top:26px">
            ${isHost ? `
              <button class="btn secondary" onclick="updateState({phase:roomData.state.returnPhase||'question'})">← Back</button>
              <button class="btn" onclick="continueFromLeaderboard()">${s.returnPhase === 'reveal' ? (isLast ? '🏆 Grand Finale →' : 'Next Question →') : 'Resume Round →'}</button>
            ` : `<span class="pill-btn">Next round coming up…</span>`}
          </div>
        </div>
      </div>`;
  }

  const track = $('#lbTrack');
  if (!track) return;
  const maxScore = Math.max(1, ...list.map(([, p]) => p.score || 0));
  track.style.height = Math.max(80, list.length * ROW_H) + 'px';

  // Phase 1: Mount rows at their previous rank (so they smoothly slide to new rank!)
  list.forEach(([id, p], newIdx) => {
    let row = lbRowNodes[id];
    const oldIdx = prevRankMap[id] !== undefined ? prevRankMap[id] : newIdx;
    const delta = oldIdx - newIdx; // positive = moved UP!
    const deltaHTML = delta > 0
      ? `<span class="lb-delta up">▲ +${delta}</span>`
      : delta < 0
        ? `<span class="lb-delta down">▼ ${delta}</span>`
        : `<span class="lb-delta same">—</span>`;

    if (!row) {
      row = lbRowNodes[id] = document.createElement('div');
      row.className = 'lb-row';
      row.style.transform = `translateY(${oldIdx * ROW_H}px)`;
      row.innerHTML = `
        <div class="lb-fill" style="width:0%"></div>
        <div class="lb-content">
          <div class="lb-rank"></div>
          <div class="lb-av"></div>
          <div class="lb-name-col"><div class="lb-name"></div><div class="lb-badges"></div></div>
          <div class="lb-delta-slot"></div>
          <div class="lb-score"></div>
        </div>`;
      track.appendChild(row);
      row.getBoundingClientRect(); // force reflow before transition
    }

    row.classList.toggle('rank-1', newIdx === 0 && (p.score || 0) > 0);
    row.classList.toggle('is-me', id === playerId);
    row.querySelector('.lb-rank').textContent = ['🥇', '🥈', '🥉'][newIdx] || `#${newIdx + 1}`;
    row.querySelector('.lb-av').textContent = p.avatar || '⚡';
    row.querySelector('.lb-name').textContent = p.name;
    row.querySelector('.lb-badges').textContent = m.rewardEmoji
      ? `${m.rewardEmoji.repeat(Math.min(p.wins || 0, 8))} ${p.wins || 0} ${m.rewardName || ''}`
      : `${p.wins || 0} correct`;
    row.querySelector('.lb-delta-slot').innerHTML = deltaHTML;
    row.querySelector('.lb-score').innerHTML = `${p.score || 0}<small> pts</small>`;

    requestAnimationFrame(() => {
      row.style.transform = `translateY(${newIdx * ROW_H}px)`;
      row.querySelector('.lb-fill').style.width = `${Math.max(8, ((p.score || 0) / maxScore) * 100)}%`;
    });
  });
}
